cmake_minimum_required(VERSION 3.16)
project(polyomino_search LANGUAGES CXX)
find_package(Threads REQUIRED)
add_executable(polyomino-search src/main.cpp)
target_compile_features(polyomino-search PRIVATE cxx_std_20)
target_link_libraries(polyomino-search PRIVATE Threads::Threads)
if(MSVC)
  target_compile_options(polyomino-search PRIVATE /W4 /permissive-)
  set_property(TARGET polyomino-search PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
  target_compile_options(polyomino-search PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  add_test(NAME invariants COMMAND polyomino-search --self-test)
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_FOUND)
    add_test(NAME cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_cli.py $<TARGET_FILE:polyomino-search>)
  endif()
endif()
